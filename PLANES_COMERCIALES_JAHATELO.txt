// Agregar a schema.prisma

enum PlanType {
  BASIC
  PREMIUM
  PLATINUM
}

model Motel {
  // ... campos existentes ...

  // NUEVO: Sistema de planes
  plan                PlanType    @default(BASIC)
  planStartDate       DateTime?
  planEndDate         DateTime?
  photoLimit          Int         @default(3)
  promoLimit          Int         @default(1)
  analyticsEnabled    Boolean     @default(false)
  reservationsEnabled Boolean     @default(false)
}

model MotelAnalytics {
  id           String   @id @default(cuid())
  motelId      String
  motel        Motel    @relation(fields: [motelId], references: [id])
  eventType    String   // 'VIEW', 'CLICK_PHONE', 'CLICK_WHATSAPP', 'CLICK_MAP'
  timestamp    DateTime @default(now())
  userLocation Json?
  source       String?  // 'MAP', 'LIST', 'SEARCH', 'HOME'

  @@index([motelId, timestamp])
}

model Reservation {
  id              String   @id @default(cuid())
  motelId         String
  motel           Motel    @relation(fields: [motelId], references: [id])
  roomTypeId      String
  roomType        RoomType @relation(fields: [roomTypeId], references: [id])
  userId          String?
  customerName    String
  customerEmail   String
  customerPhone   String
  checkIn         DateTime
  checkOut        DateTime
  status          String   @default("PENDING") // PENDING, CONFIRMED, CANCELLED
  totalAmount     Float
  commission      Float
  paymentMethod   String
  paymentStatus   String   @default("PENDING")
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  @@index([motelId, status])
  @@index([checkIn])
}
